#!/usr/bin/env python3
import random
import os

def generate_test_10(filename="tests/10_random_churn_memory_stress.txt", seed=2183):
    os.makedirs(os.path.dirname(filename), exist_ok=True)
    
    KEYS = 500
    RANDOM_COMMANDS = 9400
    
    rng = random.Random(seed)
    lines = [
        "# Target: random churn over keys 1 to 500 (dense collisions, absent deletes), "
        "then drain every key to check for memory leaks."
    ]

    live_keys = set()
    
    for i in range(1, RANDOM_COMMANDS + 1):
        # Removed the if i % 1000 == 0: X block entirely
        op = rng.choices("IDS", weights=(50, 30, 20))[0]
        key = rng.randint(1, KEYS)
        
        if op == "I":
            lines.append(f"I {key} {rng.randint(-10**6, 10**6)}")
            live_keys.add(key)
        elif op == "D":
            lines.append(f"D {key}")
            live_keys.discard(key)
        else:
            lines.append(f"S {key}")

    print(f"Sanity Check: {len(live_keys)} keys alive at the end of random phase.")

    # Enforce strict project limits
    commands = [line for line in lines if not line.startswith("#")]
